} } catch (error) { console.error("Media download error:", error); await zk.sendMessage(dest, { text: `❌ Error during ${type} download. ${error.message}` }, { quoted: originalMsg }); } } // ---------- ENHANCED URL EXTRACTION ---------- function extractDownloadUrl(data) { if (!data) return null; // Extended list of possible paths (dot notation) const paths = [ 'download_url', 'url', 'link', 'file', 'result.url', 'result.download_url', 'result.link', 'data.url', 'data.download_url', 'audio_url', 'video_url', 'result.audio_url', 'result.video_url', 'dl_link', 'response.url', 'response.download_url' ]; for (const path of paths) { const value = getNestedValue(data, path); if (value && typeof value === 'string' && value.startsWith('http')) { return value; } } // If dot notation fails, recursively search for any http string return findHttpUrl(data); } function getNestedValue(obj, path) { return path.split('.').reduce((current, key) => current?.[key], obj); } function findHttpUrl(obj) { if (!obj || typeof obj !== 'object') return null; if (Array.isArray(obj)) { for (const item of obj) { const url = findHttpUrl(item); if (url) return url; } return null; } for (const value of Object.values(obj)) { if (typeof value === 'string' && value.startsWith('http')) { return value; } if (typeof value === 'object') { const nestedUrl = findHttpUrl(value); if (nestedUrl) return nestedUrl; } } return null; }
